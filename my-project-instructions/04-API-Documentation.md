## 1. AUTHENTICATION

### Sign In
```javascript
const { data, error } = await supabase.auth.signInWithPassword({
  email: 'user@example.com',
  password: 'password123'
})
// Returns: { user, session }
```

### Sign Out
```javascript
const { error } = await supabase.auth.signOut()
```

### Get Current User
```javascript
const { data: { user } } = await supabase.auth.getUser()
```

### Reset Password
```javascript
const { error } = await supabase.auth.resetPasswordForEmail(email)
```

## 2. USER MANAGEMENT (Admin Only)

### List Users
```javascript
const { data, error } = await supabase
  .from('users')
  .select('*')
  .eq('is_active', true)
  .order('full_name')
```

### Get User by ID
```javascript
const { data, error } = await supabase
  .from('users')
  .select('*')
  .eq('id', userId)
  .single()
```

### Create User (via Edge Function)
```javascript
const { data } = await supabase.functions.invoke('create-user', {
  body: { email, password, full_name, role, permissions }
})
```

### Update User
```javascript
const { error } = await supabase
  .from('users')
  .update({ full_name, role, permissions, is_active })
  .eq('id', userId)
```

### Deactivate User
```javascript
const { error } = await supabase
  .from('users')
  .update({ is_active: false })
  .eq('id', userId)
```

## 3. CASE MANAGEMENT

### List All Cases (with related data)
```javascript
const { data, error } = await supabase
  .from('cases')
  .select(`
    *,
    case_uploads (*),
    in_depth_stages (*),
    enforcement_stages (*),
    destruction_stages (*),
    invoices (*)
  `)
  .order('case_submitted_date', { ascending: false })
```

### List Cases with Filters
```javascript
const { data, error } = await supabase
  .from('cases')
  .select('*, case_uploads(*)')
  .eq('case_status', 'APPROVED')         // Filter by status
  .eq('province', 'Ontario')              // Filter by province
  .ilike('brand_name', '%brand%')         // Search brand name
  .gte('case_submitted_date', '2024-01-01')
  .lte('case_submitted_date', '2024-12-31')
  .order('case_submitted_date', { ascending: false })
  .range(0, 49)  // Pagination: first 50 records
```

### List Cases by Status
```javascript
// In-Hand Cases
.eq('case_status', 'IN_HAND')

// Uploaded Cases
.eq('case_status', 'UPLOADED')

// Approved Cases
.eq('case_status', 'APPROVED')

// Rejected Cases
.eq('case_status', 'REJECTED')

// Closed Cases
.eq('case_status', 'CLOSED')
```

### List Cases by Client
```javascript
const { data, error } = await supabase
  .from('cases')
  .select('*, case_uploads!inner(*)')
  .eq('case_uploads.client', 'ONEWORLD')
```

### Get Single Case (Full Details)
```javascript
const { data, error } = await supabase
  .from('cases')
  .select(`
    *,
    case_uploads (*),
    in_depth_stages (*),
    enforcement_stages (*),
    final_reports (*),
    invoices (*),
    destruction_stages (*),
    case_notes (*, author:users(full_name))
  `)
  .eq('id', caseId)
  .single()
```

### Create Case
```javascript
const { data, error } = await supabase
  .from('cases')
  .insert({
    // case_id is auto-generated by trigger
    target_name: 'Target Corp',
    target_category: 'Retail',
    target_address: '123 Main St',
    city: 'Toronto',
    province: 'Ontario',
    brand_name: 'BrandX',
    products_name: 'Product A, Product B',
    case_submitted_date: '2024-01-15',
    case_submitted_by: 'John Smith',
    notes_description: 'Optional notes',
    created_by: currentUserId
  })
  .select()
  .single()
```

### Update Case
```javascript
const { error } = await supabase
  .from('cases')
  .update({
    target_name: 'Updated Name',
    notes_description: 'Updated notes'
  })
  .eq('id', caseId)
```

## 4. CASE UPLOADS (Step 2)

### Upload Case to Client
```javascript
const { data, error } = await supabase
  .from('case_uploads')
  .insert({
    case_id: caseUuid,
    client: 'ONEWORLD',  // 'ONEWORLD' | 'AA' | 'SAFEMARK'
    upload_date: '2024-01-20',
    our_fee_usd: 1500.00,
    created_by: currentUserId
  })
  .select()
  .single()
// Note: Trigger automatically updates case status to 'UPLOADED'
```

### Record Client Decision
```javascript
const { error } = await supabase
  .from('case_uploads')
  .update({
    decision_status: 'APPROVED',  // 'APPROVED' | 'REJECTED'
    decision_date: '2024-01-25'
  })
  .eq('case_id', caseId)
// Note: Trigger updates case status and creates workflow stages if approved
```

## 5. WORKFLOW STAGES

### Create In-Depth Stage
```javascript
const { data, error } = await supabase
  .from('in_depth_stages')
  .insert({
    case_id: caseUuid,
    target_date: '2024-02-01',  // approval_date + 7 days
    status: 'IN_PROGRESS',
    created_by: currentUserId
  })
  .select()
  .single()
```

### Update In-Depth Stage
```javascript
const { error } = await supabase
  .from('in_depth_stages')
  .update({
    target_date: '2024-02-05',  // If changing date
    status: 'DONE',
    status_date: new Date().toISOString(),
    notes: 'Completed successfully'
  })
  .eq('case_id', caseId)
```

### Create Enforcement Stage
```javascript
const { data, error } = await supabase
  .from('enforcement_stages')
  .insert({
    case_id: caseUuid,
    target_date: '2024-02-08',  // in_depth_target + 7 days
    status: 'IN_PROGRESS',
    created_by: currentUserId
  })
  .select()
  .single()
```

### Update Enforcement Stage
```javascript
const { error } = await supabase
  .from('enforcement_stages')
  .update({
    status: 'DONE',
    status_date: new Date().toISOString(),
    notes: 'Enforcement complete'
  })
  .eq('case_id', caseId)
```

### Create Final Report
```javascript
const { data, error } = await supabase
  .from('final_reports')
  .insert({
    case_id: caseUuid,
    submission_date: '2024-02-10',
    created_by: currentUserId
  })
  .select()
  .single()
```

### Create Destruction Stage
```javascript
const { data, error } = await supabase
  .from('destruction_stages')
  .insert({
    case_id: caseUuid,
    due_date: '2024-02-09',  // day after enforcement done
    status: 'IN_PROGRESS',
    created_by: currentUserId
  })
  .select()
  .single()
```

### Complete Destruction (Auto-Closes Case)
```javascript
const { error } = await supabase
  .from('destruction_stages')
  .update({
    status: 'DONE',
    status_date: new Date().toISOString()
  })
  .eq('case_id', caseId)
// Note: Trigger automatically sets case status to 'CLOSED' and sets closed_date
```

### List Workflow Stages by Status
```javascript

  .from('in_depth_stages')
  .select('*, cases(*)')
  .eq('status', 'IN_PROGRESS')
  .order('target_date', { ascending: true })

// Enforcement Done
const { data } = await supabase
  .from('enforcement_stages')
  .select('*, cases(*)')
  .eq('status', 'DONE')
  .order('status_date', { ascending: false })
```

## 6. INVOICES

### Create Invoice
```javascript
const { data, error } = await supabase
  .from('invoices')
  .insert({
    case_id: caseUuid,
    // invoice_number is auto-generated
    issue_date: '2024-02-15',
    due_date: '2024-04-15',  // issue_date + 60 days
    status: 'ISSUED',
    created_by: currentUserId
  })
  .select()
  .single()
```

### Update Invoice Status
```javascript
const { error } = await supabase
  .from('invoices')
  .update({
    status: 'PAID',  // 'ISSUED' | 'PAID' | 'NOT_PAID'
    status_date: new Date().toISOString()
  })
  .eq('case_id', caseId)
```

### List Invoices by Status
```javascript
// Unpaid Invoices
const { data } = await supabase
  .from('invoices')
  .select('*, cases(*, case_uploads(*))')
  .in('status', ['ISSUED', 'NOT_PAID'])
  .order('due_date', { ascending: true })

// Paid Invoices
const { data } = await supabase
  .from('invoices')
  .select('*, cases(*, case_uploads(*))')
  .eq('status', 'PAID')
  .order('status_date', { ascending: false })
```

### List Overdue Invoices
```javascript
const { data } = await supabase
  .from('invoices')
  .select('*, cases(*, case_uploads(*))')
  .in('status', ['ISSUED', 'NOT_PAID'])
  .lt('due_date', new Date().toISOString().split('T')[0])
  .order('due_date', { ascending: true })
```

## 11. COMMON QUERY PATTERNS

### Pagination
```javascript
const pageSize = 20
const page = 0  // 0-indexed

const { data, count, error } = await supabase
  .from('cases')
  .select('*', { count: 'exact' })
  .range(page * pageSize, (page + 1) * pageSize - 1)
```

### Search
```javascript
const { data, error } = await supabase
  .from('cases')
  .select('*')
  .or(`target_name.ilike.%${search}%,brand_name.ilike.%${search}%,case_id.ilike.%${search}%`)
```

### Date Range Filtering
```javascript
const { data, error } = await supabase
  .from('cases')
  .select('*')
  .gte('case_submitted_date', startDate)
  .lte('case_submitted_date', endDate)
```

### Multiple Status Filter
```javascript
const { data, error } = await supabase
  .from('cases')
  .select('*')
  .in('case_status', ['APPROVED', 'UPLOADED'])
```

### Count Records
```javascript
const { count, error } = await supabase
  .from('cases')
  .select('*', { count: 'exact', head: true })
  .eq('case_status', 'IN_HAND')
```

## 12. ERROR HANDLING

```javascript
const { data, error } = await supabase
  .from('cases')
  .select('*')

if (error) {
  console.error('Error:', error.message)
  // Handle specific error codes
  if (error.code === 'PGRST116') {
    // No rows returned for .single()
  }
  if (error.code === '42501') {
    // Permission denied (RLS policy)
  }
  return
}

// Success - use data
console.log(data)
```

## 13. REALTIME SUBSCRIPTIONS (Optional)

### Subscribe to Case Changes
```javascript
const subscription = supabase
  .channel('cases-changes')
  .on(
    'postgres_changes',
    { event: '*', schema: 'public', table: 'cases' },
    (payload) => {
      console.log('Change received:', payload)
    }
  )
  .subscribe()

// Cleanup
subscription.unsubscribe()
```
